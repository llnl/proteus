121inline unsigned parseTraceConfig(const char *VarName) {
122 const char *EnvValue = std::getenv(VarName);
123 if (!EnvValue)
124 return 0;
125
126 unsigned Mask = 0;
127 std::istringstream Stream(EnvValue);
128 std::string Token;
129 while (std::getline(Stream, Token, ';')) {
130 if (Token.empty())
131 continue;
132 if (Token == "specialization")
133 Mask |= static_cast<unsigned>(TraceOption::Specialization);
134 else if (Token == "ir-dump")
135 Mask |= static_cast<unsigned>(TraceOption::IRDump);
136 else if (Token == "kernel-trace")
137 Mask |= static_cast<unsigned>(TraceOption::KernelTrace);
138 else if (Token == "cache-stats")
139 Mask |= static_cast<unsigned>(TraceOption::CacheStats);
140 else
141 reportFatalError("Unknown PROTEUS_TRACE_OUTPUT token: " + Token);
142 }
143 return Mask;
144}

187 static std::optional<LaunchDim> getTunedDim(const llvm::json::Object &Config,
188 llvm::StringRef Key) {
189 const auto *Value = Config.get(Key);
190 if (!Value)
191 return std::nullopt;
192
193 const auto *Object = Value->getAsObject();
194 if (!Object)
195 reportFatalError("Invalid tuned " + Key.str() +
196 ": expected an object with x, y, z");
197
198 LaunchDim Dim;
199 const char *Axes[] = {"x", "y", "z"};
200 for (unsigned I = 0; I < Dim.size(); ++I) {
201 const char *Axis = Axes[I];
202 auto Coordinate = Object->getInteger(Axis);
203 if (!Coordinate || *Coordinate <= 0 ||
204 static_cast<uint64_t>(*Coordinate) >
205 std::numeric_limits<unsigned int>::max())
206 reportFatalError("Invalid tuned " + Key.str() + "." + Axis +
207 ": expected a positive 32-bit integer");
208 Dim[I] = static_cast<unsigned int>(*Coordinate);
209 }
210 return Dim;
211 }

344inline llvm::StringMap<const CodeGenerationConfig>
345parseJSONConfig(std::optional<std::string> JSONFn) {
346 llvm::StringMap<const CodeGenerationConfig> TunedConfigs;
347 if (!JSONFn)
348 return TunedConfigs;
349
350 auto JSONRoot = [&JSONFn]() -> std::optional<llvm::json::Object> {
351 llvm::ErrorOr<std::unique_ptr<llvm::MemoryBuffer>> JSONBuf =
352 llvm::MemoryBuffer::getFile(JSONFn.value(), /* isText */ true,
353 /* RequiresNullTerminator */ true);
354 if (!JSONBuf)
355 reportFatalError("Error when opening json file: " +
356 JSONBuf.getError().message() + "\n");
357
358 llvm::json::Value JsonInfo =
359 llvm::cantFail(llvm::json::parse(JSONBuf.get()->getBuffer()),
360 "Cannot convert buffer to json value");
361
362 if (auto *Obj = JsonInfo.getAsObject())
363 return *Obj;
364 return std::nullopt;
365 }();
366
367 if (!JSONRoot)
368 reportFatalError("Top-level JSON is not an object.\n");
369
370 for (auto &KV : JSONRoot.value()) {
371 auto KernelName = KV.first;
372 if (const auto *Options = KV.second.getAsObject()) {
373 TunedConfigs.try_emplace(
374 KernelName, CodeGenerationConfig::createFromJSONEntry(*Options));
375 }
376 }
377 return TunedConfigs;
378}
