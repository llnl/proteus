69inline Expected<std::unique_ptr<TargetMachine>>
70createTargetMachine(Module &M, StringRef Arch, unsigned OptLevel = 3) {
71 Triple TT(M.getTargetTriple());
72 auto CGOptLevel = CodeGenOpt::getLevel(OptLevel);
73 if (CGOptLevel == std::nullopt)
74 reportFatalError("Invalid opt level");
75
76 std::string Msg;
77 const Target *T = TargetRegistry::lookupTarget(M.getTargetTriple(), Msg);
78 if (!T)
79 return make_error<StringError>(Msg, inconvertibleErrorCode());
80
81 SubtargetFeatures Features;
82 Features.getDefaultSubtargetFeatures(TT);
83
84 std::optional<Reloc::Model> RelocModel;
85 if (M.getModuleFlag("PIC Level"))
86 RelocModel =
87 M.getPICLevel() == PICLevel::NotPIC ? Reloc::Static : Reloc::PIC_;
88
89 std::optional<CodeModel::Model> CodeModel = M.getCodeModel();
90
91 // Use default target options.
92 // TODO: Customize based on AOT compilation flags or by creating a
93 // constructor that sets target options based on the triple.
94 TargetOptions Options;
95 std::unique_ptr<TargetMachine> TM(T->createTargetMachine(
96 M.getTargetTriple(), Arch, Features.getString(), Options, RelocModel,
97 CodeModel, CGOptLevel.value()));
98 if (!TM)
99 return make_error<StringError>("Failed to create target machine",
100 inconvertibleErrorCode());
101 return TM;
102}

132inline std::string composeOptimizationPassPipeline(
133 std::optional<std::string> PassPipeline, char OptLevel,
134 const std::vector<JITPassPluginConfig> &Plugins) {
135 std::string Pipeline;
136 for (const auto &Plugin : Plugins) {
137 if (!Plugin.Insertion ||
138 Plugin.Insertion->Position != JITPassPluginPosition::Prepend)
139 continue;
140 if (!Pipeline.empty())
141 Pipeline += ",";
142 Pipeline += Plugin.Insertion->Pipeline;
143 }
144
145 if (!Pipeline.empty())
146 Pipeline += ",";
147 Pipeline += PassPipeline ? std::move(*PassPipeline)
148 : getDefaultOptimizationPipeline(OptLevel);
149
150 for (const auto &Plugin : Plugins) {
151 if (!Plugin.Insertion ||
152 Plugin.Insertion->Position != JITPassPluginPosition::Append)
153 continue;
154 Pipeline += ",";
155 Pipeline += Plugin.Insertion->Pipeline;
156 }
157
158 return Pipeline;
159}

161// LLVM 23 removed the textual Os/Oz pipeline names. Keep accepting these
162// names in PROTEUS_OPT_PIPELINE for compatibility with existing callers, but
163// translate them only at the point where LLVM parses the pipeline. The
164// caller-facing trace can therefore continue to report the requested
165// pipeline.
166inline std::string normalizeOptimizationPassPipeline(StringRef PassPipeline) {
167#if LLVM_VERSION_MAJOR > 22
168 std::string Normalized = PassPipeline.str();
169 for (auto [Old, New] :
170 {std::pair<StringRef, StringRef>{"default<Os>", "default<O2>"},
171 {"default<Oz>", "default<O2>"}}) {
172 size_t Pos = 0;
173 while ((Pos = Normalized.find(Old.str(), Pos)) != std::string::npos) {
174 Normalized.replace(Pos, Old.size(), New.str());
175 Pos += New.size();
176 }
177 }
178 return Normalized;
179#else
180 return PassPipeline.str();
181#endif
182}

238inline void runOptimizationPassPipeline(
239 Module &M, StringRef Arch, const std::string &PassPipeline,
240 unsigned CodegenOptLevel,
241 const std::vector<JITPassPluginConfig> &Plugins = {}) {
242 PipelineTuningOptions PTO;
243
244 std::optional<PGOOptions> PGOOpt;
245 auto TM = createTargetMachine(M, Arch, CodegenOptLevel);
246 if (auto Err = TM.takeError())
247 report_fatal_error(std::move(Err));
248 TargetLibraryInfoImpl TLII(Triple(M.getTargetTriple()));
249
250 auto LoadedPlugins = loadJITPassPlugins(Plugins);
251 PassBuilder PB(TM->get(), PTO, PGOOpt, nullptr);
252 for (const auto &Plugin : LoadedPlugins)
253 Plugin.registerPassBuilderCallbacks(PB);
254 LoopAnalysisManager LAM;
255 FunctionAnalysisManager FAM;
256 CGSCCAnalysisManager CGAM;
257 ModuleAnalysisManager MAM;
258
259 FAM.registerPass([&] { return TargetLibraryAnalysis(TLII); });
260
261 PB.registerModuleAnalyses(MAM);
262 PB.registerCGSCCAnalyses(CGAM);
263 PB.registerFunctionAnalyses(FAM);
264 PB.registerLoopAnalyses(LAM);
265 PB.crossRegisterProxies(LAM, FAM, CGAM, MAM);
266 ModulePassManager Passes;
267#if LLVM_VERSION_MAJOR > 22
268 // Preserve the pre-LLVM 23 Os/Oz semantics while normalizing their removed
269 // textual pipelines to O2 below.
270 applySizeOptimizationAttribute(M, PassPipeline);
271#endif
272 const std::string NormalizedPipeline =
273 normalizeOptimizationPassPipeline(PassPipeline);
274 if (auto E = PB.parsePassPipeline(Passes, NormalizedPipeline))
275 reportFatalError("Error: " + toString(std::move(E)));
276
277 Passes.run(M, MAM);
278}

280inline void runOptimizationPassPipeline(
281 Module &M, StringRef Arch, char OptLevel = '3',
282 unsigned CodegenOptLevel = 3,
283 const std::vector<JITPassPluginConfig> &Plugins = {}) {
284 PipelineTuningOptions PTO;
285
286 std::optional<PGOOptions> PGOOpt;
287 auto TM = createTargetMachine(M, Arch, CodegenOptLevel);
288 if (auto Err = TM.takeError())
289 report_fatal_error(std::move(Err));
290 TargetLibraryInfoImpl TLII(Triple(M.getTargetTriple()));
291
292 auto LoadedPlugins = loadJITPassPlugins(Plugins);
293 PassBuilder PB(TM->get(), PTO, PGOOpt, nullptr);
294 for (const auto &Plugin : LoadedPlugins)
295 Plugin.registerPassBuilderCallbacks(PB);
296 LoopAnalysisManager LAM;
297 FunctionAnalysisManager FAM;
298 CGSCCAnalysisManager CGAM;
299 ModuleAnalysisManager MAM;
300
301 FAM.registerPass([&] { return TargetLibraryAnalysis(TLII); });
302
303 PB.registerModuleAnalyses(MAM);
304 PB.registerCGSCCAnalyses(CGAM);
305 PB.registerFunctionAnalyses(FAM);
306 PB.registerLoopAnalyses(LAM);
307 PB.crossRegisterProxies(LAM, FAM, CGAM, MAM);
308
309 OptimizationLevel OptSetting;
310#if LLVM_VERSION_MAJOR > 22
311 // Preserve the pre-LLVM 23 Os/Oz semantics with O2 plus function attrs.
312 applySizeOptimizationAttribute(M, OptLevel);
313#endif
314 switch (OptLevel) {
315 case '0':
316 OptSetting = OptimizationLevel::O0;
317 break;
318 case '1':
319 OptSetting = OptimizationLevel::O1;
320 break;
321 case '2':
322 OptSetting = OptimizationLevel::O2;
323 break;
324 case '3':
325 OptSetting = OptimizationLevel::O3;
326 break;
327 case 's':
328#if LLVM_VERSION_MAJOR > 22
329 // LLVM 23 uses O2 plus the optsize function attribute.
330 OptSetting = OptimizationLevel::O2;
331#else
332 OptSetting = OptimizationLevel::Os;
333#endif
334 break;
335 case 'z':
336#if LLVM_VERSION_MAJOR > 22
337 // LLVM 23 uses O2 plus the minsize function attribute.
338 OptSetting = OptimizationLevel::O2;
339#else
340 OptSetting = OptimizationLevel::Oz;
341#endif
342 break;
343 default:
344 reportFatalError(std::string("Unsupported optimization level ") + OptLevel);
345 };
346
347 ModulePassManager Passes = PB.buildPerModuleDefaultPipeline(OptSetting);
348 Passes.run(M, MAM);
349}

441inline void pruneDanglingNVVMAnnotations(Module &M) {
442 NamedMDNode *Annotations = M.getNamedMetadata("nvvm.annotations");
443 if (!Annotations)
444 return;
445
446 SmallVector<MDNode *> LiveEntries;
447 for (MDNode *Entry : Annotations->operands()) {
448 // Global DCE nulls out the entry of a function it removes and stripping
449 // debug info rewrites such a node to an empty tuple, !{}.
450 // LLVM's nvvm.annotations which LLVM's later UpgradeNVVMAnnotations
451 // pass reads unconditionally, so we need to prune out such entries.
452 // Otherwise, UpgradeNVVMAnnotations runs out of bounds!
453 if (!Entry || Entry->getNumOperands() == 0)
454 continue;
455
456 if (!mdconst::dyn_extract_or_null<GlobalValue>(Entry->getOperand(0)))
457 continue;
458
459 LiveEntries.push_back(Entry);
460 }
461
462 if (LiveEntries.size() == Annotations->getNumOperands())
463 return;
464
465 Annotations->clearOperands();
466 for (MDNode *Entry : LiveEntries)
467 Annotations->addOperand(Entry);
468}

540inline void pruneIR(Module &M, bool UnsetExternallyInitialized = true) {
541 // Remove llvm.global.annotations now that we have read them.
542 if (auto *GlobalAnnotations = M.getGlobalVariable("llvm.global.annotations"))
543 M.eraseGlobalVariable(GlobalAnnotations);
544
545 // Remove llvm.compiler.used
546 if (auto *CompilerUsed = M.getGlobalVariable("llvm.compiler.used"))
547 M.eraseGlobalVariable(CompilerUsed);
548
549 // Remove the __clang_gpu_used_external used in HIP RDC compilation and its
550 // uses in llvm.used, llvm.compiler.used.
551 SmallVector<GlobalVariable *> GlobalsToErase;
552 for (auto &GV : M.globals()) {
553 auto Name = GV.getName();
554 if (Name.starts_with("__clang_gpu_used_external") ||
555 Name.starts_with("_jit_bitcode") || Name.starts_with("__hip_cuid")) {
556 GlobalsToErase.push_back(&GV);
557 removeFromUsedLists(M, [&GV](Constant *C) {
558 if (auto *Global = dyn_cast<GlobalVariable>(C))
559 return Global == &GV;
560 return false;
561 });
562 }
563 }
564 for (auto *GV : GlobalsToErase) {
565 M.eraseGlobalVariable(GV);
566 }
567
568 // Remove externaly_initialized attributes.
569 if (UnsetExternallyInitialized)
570 for (auto &GV : M.globals())
571 if (GV.isExternallyInitialized())
572 GV.setExternallyInitialized(false);
573}
