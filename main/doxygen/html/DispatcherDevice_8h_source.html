131 // Skipping optimization on the RTC path leaves the runtime compiler in charge

133 // Warn once per process so the setting does not appear to take effect.
134 static void
135 warnOptimizationConfigIgnoredByRTC(const CodeGenerationConfig &CGConfig) {
136 // '3' and 3 are the PROTEUS_OPT_LEVEL and PROTEUS_CODEGEN_OPT_LEVEL
137 // defaults set by CodeGenerationConfig.
138 const bool UsesDefaults =
139 !CGConfig.optPipeline() && CGConfig.optLevel() == '3' &&
140 CGConfig.codeGenOptLevel() == 3 && getJITPassPluginConfigs().empty();
141 if (UsesDefaults)
142 return;
143
144 static std::once_flag WarnOnce;
145 std::call_once(WarnOnce, [] {
146 Logger::outs("proteus")
147 << "Warning: RTC codegen optimizes internally, so Proteus ignores "
148 "PROTEUS_OPT_PIPELINE, PROTEUS_OPT_LEVEL, "
149 "PROTEUS_CODEGEN_OPT_LEVEL and JIT pass plugins, use "
150 "PROTEUS_CODEGEN=serial or PROTEUS_CODEGEN=parallel to apply "
