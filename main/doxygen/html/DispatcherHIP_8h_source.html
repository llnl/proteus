35 auto LoadBitcode = [&](const llvm::SmallString<256> &Path) {
36 auto BufferOrErr = llvm::MemoryBuffer::getFile(Path);

41 BufferOrErr->get()->getMemBufferRef(), M.getContext());

44 Path.str().str() + " (" + Toolchain.Origin + ")");
