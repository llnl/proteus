100inline HashT hashArrayRefElement(const RuntimeConstant &RC) {
101 if (RC.Type == RuntimeConstantType::ARRAY ||
102 RC.Type == RuntimeConstantType::STATIC_ARRAY ||
103 RC.Type == RuntimeConstantType::VECTOR) {
104 switch (RC.ArrInfo.EltType) {
105 case RuntimeConstantType::BOOL:
106 return hashRuntimeConstantArray<bool>(RC);
107 case RuntimeConstantType::INT8:
108 return hashRuntimeConstantArray<int8_t>(RC);
109 case RuntimeConstantType::INT32:
110 return hashRuntimeConstantArray<int32_t>(RC);
111 case RuntimeConstantType::INT64:
112 return hashRuntimeConstantArray<int64_t>(RC);
113 case RuntimeConstantType::FLOAT:
114 return hashRuntimeConstantArray<float>(RC);
115 case RuntimeConstantType::DOUBLE:
116 return hashRuntimeConstantArray<double>(RC);
117 default:
118 reportFatalError("Unsupported array element type: " +
119 toString(RC.ArrInfo.EltType));
120 }
121 } else if (RC.Type == RuntimeConstantType::OBJECT) {
122 return hashRuntimeConstantObject(RC);
123 } else if (isScalarRuntimeConstantType(RC.Type)) {
124 return hashValue(
125 StringRef{reinterpret_cast<const char *>(&RC.Value), sizeof(RC.Value)});
126 }
127
128 reportFatalError("Unsupported type " + toString(RC.Type));
129}

131inline HashT hashValue(const RuntimeConstant &RC) {
132 HashT H = hashValue(static_cast<int32_t>(RC.Type));
133 H = stable_hash_combine(H.getValue(), hashValue(RC.Pos).getValue());
134 H = stable_hash_combine(H.getValue(), hashValue(RC.Offset).getValue());
135
136 if (RC.Type == RuntimeConstantType::ARRAY ||
137 RC.Type == RuntimeConstantType::STATIC_ARRAY ||
138 RC.Type == RuntimeConstantType::VECTOR) {
139 H = stable_hash_combine(H.getValue(),
140 hashValue(RC.ArrInfo.NumElts).getValue());
141 H = stable_hash_combine(
142 H.getValue(),
143 hashValue(static_cast<int32_t>(RC.ArrInfo.EltType)).getValue());
144 H = stable_hash_combine(H.getValue(), hashArrayRefElement(RC).getValue());
145 return H;
146 }
147
148 if (RC.Type == RuntimeConstantType::OBJECT) {
149 H = stable_hash_combine(H.getValue(),
150 hashValue(RC.ObjInfo.Size).getValue());
151 H = stable_hash_combine(
152 H.getValue(),
153 hashValue(static_cast<int32_t>(RC.ObjInfo.PassByValue)).getValue());
154 H = stable_hash_combine(H.getValue(), hashArrayRefElement(RC).getValue());
155 return H;
156 }
157
158 if (isScalarRuntimeConstantType(RC.Type)) {
159 H = stable_hash_combine(H.getValue(), hashArrayRefElement(RC).getValue());
160 return H;
161 }
162
163 // For NONE/unsupported values, the type/pos/offset hash above is all we keep.
164 return H;
165}

167inline HashT hashValue(ArrayRef<RuntimeConstant> Arr) {
168 if (Arr.empty())
169 return 0;
170
171 HashT HashValue = hashArrayRefElement(Arr[0]);
172 for (int I = 1, E = Arr.size(); I < E; ++I)
173 HashValue = stable_hash_combine(HashValue.getValue(),
174 hashArrayRefElement(Arr[I]).getValue());
175
176 return HashValue;
177}

195template <typename Key, typename Value>
196inline std::enable_if_t<
197 std::is_integral<Key>::value || std::is_enum<Key>::value, HashT>
198hashValue(const DenseMap<Key, Value> &Map) {
199 if (Map.empty())
200 return 0;
201
202 SmallVector<Key, 32> Keys;
203 Keys.reserve(Map.size());
204 for (const auto &KV : Map)
205 Keys.push_back(KV.first);
206
207 std::sort(Keys.begin(), Keys.end());
208
209 HashT H = hashValue(static_cast<uint64_t>(Map.size()));
210 for (Key K : Keys) {
211 auto It = Map.find(K);
212 if (It == Map.end())
213 reportFatalError("Internal error: DenseMap key vanished during hashing");
214
215 HashT PairH = stable_hash_combine(hashValue(K).getValue(),
216 hashValue(It->second).getValue());
217 H = stable_hash_combine(H.getValue(), PairH.getValue());
218 }
219
220 return H;
221}

232inline HashT hashCodeGenConfig(const CodeGenerationConfig &CGConfig) {
233 HashT H = hashValue(static_cast<int>(CGConfig.codeGenOption()));
234 H = hashCombine(H, hashValue(CGConfig.optLevel()));
235 H = hashCombine(H, hashValue(CGConfig.codeGenOptLevel()));
236 if (auto Pipeline = CGConfig.optPipeline())
237 H = hashCombine(H, hashValue(Pipeline.value()));
238 for (const auto &Plugin : getJITPassPluginConfigs()) {
239 H = hashCombine(H, hashValue(Plugin.Path));
240 H = hashCombine(H, hashValue(Plugin.Insertion.has_value()));
241 if (Plugin.Insertion) {
242 H = hashCombine(H, hashValue(Plugin.Insertion->Pipeline));
243 H = hashCombine(H,
244 hashValue(static_cast<int>(Plugin.Insertion->Position)));
245 }
246 H = hashCombine(H, hashValue(Plugin.Fingerprint));
247 }
248 return H;
249}
