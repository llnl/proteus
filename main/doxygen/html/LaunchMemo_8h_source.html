34// Recent specializations of one kernel, keyed by the raw launch inputs that
35// feed the specialization hash. Keys have a fixed per-kernel size.
36class LaunchMemo {
37public:
38 static constexpr unsigned NumSlots = 4;
39
40 // Creates a disabled memo.
41 LaunchMemo() = default;
42
43 // Creates a memo, disabled if any runtime constant is not a scalar.
44 LaunchMemo(ArrayRef<RuntimeConstantInfo *> RCInfoArray) {
45 if (!all_of(RCInfoArray, [](const RuntimeConstantInfo *RCInfo) {
46 return isScalarRuntimeConstantType(RCInfo->ArgInfo.Type);
47 }))
48 return;
49
50 PrefixLen = 2 * sizeof(dim3) + sizeof(uint64_t) + CGConfigSize;
51 for (const auto *RCInfo : RCInfoArray) {
52 RuntimeConstantType Type = RCInfo->ArgInfo.Type;
53 uint32_t Size = Type == RuntimeConstantType::PTR ? sizeof(void *)
54 : getSizeInBytes(Type);
55 Fields.push_back({static_cast<uint32_t>(RCInfo->ArgInfo.Pos), Size});
56 PrefixLen += Size;
57 }
58 Enabled = true;
59 }
60
61 LaunchMemo(LaunchMemo &&) = default;
62 LaunchMemo &operator=(LaunchMemo &&) = default;
63
64 bool isEnabled() const { return Enabled; }
65
66 // Serializes the launch inputs into Key.
67 void encode(SmallVectorImpl<char> &Key, void **KernelArgs, dim3 GridDim,
68 dim3 BlockDim, uint64_t PluginGeneration,
69 const CodeGenerationConfig &CGConfig,
70 const LambdaRegistry::DeviceLaunchInfo &LaunchInfo) const {
71 assert(Enabled && "Expected enabled launch memo");
72
73 size_t Len = PrefixLen;
74 for (const auto &[CallsiteIndex, RCs] : LaunchInfo.CallsiteRuntimeConstants)
75 Len += sizeof(uint64_t) + RCs.size() * LambdaRCSize;
76 if (KeyLen != 0 && Len != KeyLen)
77 reportFatalError("Launch memo key size changed");
78
79 Key.resize_for_overwrite(Len);
80
81 char *Out = Key.data();
82 for (const ArgField &F : Fields)
83 Out = putField(Out, KernelArgs[F.Pos], F.Size);
84 Out = put(Out, BlockDim);
85 bool KeyGridDim =
86 CGConfig.specializeDims() || CGConfig.specializeDimsRange();
87 Out = put(Out, KeyGridDim ? GridDim : dim3{0, 0, 0});
88 Out = put(Out, PluginGeneration);
89 Out = putCGConfig(Out, CGConfig);
90 for (const auto &[CallsiteIndex, RCs] :
91 LaunchInfo.CallsiteRuntimeConstants) {
92 Out = put(Out, static_cast<uint64_t>(CallsiteIndex));
93 for (const auto &RC : RCs) {
94 assert(isScalarRuntimeConstantType(RC.Type) &&
95 "Expected scalar lambda runtime constant");
96 Out = put(Out, RC.Type);
97 Out = put(Out, RC.Pos);
98 Out = put(Out, RC.Offset);
99 Out = put(Out, RC.Value);
100 }
101 }
102 assert(Out == Key.data() + Len && "Launch memo key size mismatch");
103 }
104
105 // Returns the memoized kernel function, or nullptr.
106 void *lookup(ArrayRef<char> Key) const {
107 for (unsigned I = 0; I < NumSlots; ++I)
108 if (Funcs[I] && std::memcmp(slot(I), Key.data(), KeyLen) == 0)
109 return Funcs[I];
110 return nullptr;
111 }
112
113 // Replaces the oldest entry.
114 void insert(ArrayRef<char> Key, void *KernelFunc) {
115 assert(Enabled && "Expected enabled launch memo");
116 if (KeyLen == 0) {
117 KeyLen = Key.size();
118 Keys.reset(new char[NumSlots * size_t(KeyLen)]);
119 } else if (Key.size() != KeyLen) {
120 reportFatalError("Launch memo key size changed");
121 }
122
123 std::memcpy(slot(Next), Key.data(), KeyLen);
124 Funcs[Next] = KernelFunc;
125 Next = (Next + 1) % NumSlots;
126 }
127
128private:
129 struct ArgField {
130 uint32_t Pos;
131 uint32_t Size;
132 };
133
134 // Type, Pos, Offset, Value.
135 static constexpr size_t LambdaRCSize = sizeof(RuntimeConstantType) +
136 2 * sizeof(int32_t) +
137 sizeof(RuntimeConstantValue);
138
139 // Pipeline hash, codegen opt level, tuned max threads, tuned min blocks per
140 // SM, codegen, opt level, flags.
141 static constexpr size_t CGConfigSize =
142 sizeof(uint64_t) + 3 * sizeof(int) + 3 * sizeof(uint8_t);
143
144 template <typename T> static char *put(char *Out, const T &V) {
145 std::memcpy(Out, &V, sizeof(T));
146 return Out + sizeof(T);
147 }
148
149 static char *putCGConfig(char *Out, const CodeGenerationConfig &CGConfig) {
150 uint8_t Flags =
151 CGConfig.specializeArgs() | CGConfig.specializeLaunchBounds() << 1 |
152 CGConfig.specializeDims() << 2 | CGConfig.specializeDimsRange() << 3;
153 Out = put(Out, CGConfig.optPipelineHash());
154 Out = put(Out, CGConfig.codeGenOptLevel());
155 Out = put(Out, CGConfig.tunedMaxThreads());
156 Out = put(Out, CGConfig.tunedMinBlocksPerSM());
157 Out = put(Out, static_cast<uint8_t>(CGConfig.codeGenOption()));
158 Out = put(Out, static_cast<uint8_t>(CGConfig.optLevel()));
159 Out = put(Out, Flags);
160 return Out;
161 }
162
163 static char *putField(char *Out, const void *In, uint32_t Size) {
164 switch (Size) {
165 case 1:
166 std::memcpy(Out, In, 1);
167 break;
168 case 4:
169 std::memcpy(Out, In, 4);
170 break;
171 case 8:
172 std::memcpy(Out, In, 8);
173 break;
174 case 16:
175 std::memcpy(Out, In, 16);
176 break;
177 default:
178 std::memcpy(Out, In, Size);
179 }
180 return Out + Size;
181 }
182
183 char *slot(unsigned I) const { return Keys.get() + size_t(I) * KeyLen; }
184
185 SmallVector<ArgField, 6> Fields;
186 // NumSlots * KeyLen, set on first insert.
187 std::unique_ptr<char[]> Keys;
188 // nullptr marks an empty slot.
189 std::array<void *, NumSlots> Funcs{};
190 // Kernel RCs, dims, plugin generation, codegen config.
191 uint32_t PrefixLen = 0;
192 // 0 until the first insert.
193 uint32_t KeyLen = 0;
194 uint8_t Next = 0;
195 bool Enabled = false;
196};
